```cpp
case TelemetryEventType::first_byte_received: {
    // D-21: Ignore if already set (duplicate event)

        first_byte_received_ns_ = event.timestamp_ns;
        uint64_t ttfb_ns = event.timestamp_ns - task_started_ns_;

        first_byte_received_set_.store(true, std::memory_order_relaxed);
    }
    break;
}
```

<task type="auto">
  <name>Task 4: Implement packet statistics in handle_event()</name>
  <files>src/telemetry/telemetry_collector.cpp</files>
  <action>
Add case handler for download_delta event to update packet statistics:

```cpp
case TelemetryEventType::download_delta: {
    size_t bytes = event.payload.download_delta.bytes;
    
    // D-09: Increment packet counter
    packets_enqueued_total_.fetch_add(1, std::memory_order_relaxed);
    
    // D-10: Update total bytes for average calculation
    total_packet_bytes_.fetch_add(bytes, std::memory_order_relaxed);
    
    // D-11: Update max packet size using CAS loop
    size_t current_max = max_packet_size_bytes_.load(std::memory_order_relaxed);
    while (bytes > current_max) {
        if (max_packet_size_bytes_.compare_exchange_weak(current_max, bytes,
                std::memory_order_relaxed, std::memory_order_relaxed)) {
            break;
        }
    }
    
    break;
}
```

The DownloadDelta payload provides `bytes` field. Use compare_exchange_weak for lock-free max update.
