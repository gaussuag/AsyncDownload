Update the existing download_delta case handler to compute EMA speed:

```cpp
case TelemetryEventType::download_delta: {
    size_t bytes = event.payload.download_delta.bytes;
    
    // Existing packet statistics...
    packets_enqueued_total_.fetch_add(1, std::memory_order_relaxed);
    total_packet_bytes_.fetch_add(bytes, std::memory_order_relaxed);
    size_t current_max = max_packet_size_bytes_.load(std::memory_order_relaxed);
    while (bytes > current_max) {
        if (max_packet_size_bytes_.compare_exchange_weak(current_max, bytes,
                std::memory_order_relaxed, std::memory_order_relaxed)) {
            break;
        }
    }
    
    // NEW: EMA speed calculation (D-04, D-05, D-06)
    if (last_network_timestamp_ns_ > 0) {
        uint64_t time_delta_ns = event.timestamp_ns - last_network_timestamp_ns_;
        if (time_delta_ns > 0) {
            double current_speed = static_cast<double>(bytes) * 1e9 / static_cast<double>(time_delta_ns);
            // EMA: EMA_new = 0.8 * EMA_old + 0.2 * current
            network_speed_ema_ = 0.8 * network_speed_ema_ + 0.2 * current_speed;
        }
    }
    last_network_timestamp_ns_ = event.timestamp_ns;
    total_download_bytes_.fetch_add(bytes, std::memory_order_relaxed);
    
    break;
}
```

per D-04: α = 0.8 means old_speed weighted 0.8, current_speed weighted 0.2.
