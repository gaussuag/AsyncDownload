#include <gtest/gtest.h>
#include "asyncdownload/telemetry/telemetry_collector.hpp"
#include "asyncdownload/telemetry/telemetry_sink.hpp"

namespace asyncdownload::telemetry {

class TelemetryCollectorTest : public ::testing::Test {
protected:
    TelemetrySink sink_;
    TelemetryCollector collector_;
    
    TelemetryCollectorTest() : collector_(sink_) {}
    
    void SetUp() override {
        collector_.start_consuming();
    }
    
    void TearDown() override {
        collector_.stop_consuming();
        collector_.wait_until_drained();
    }
};

TEST_F(TelemetryCollectorTest, TTFB_Calculation) {
    // Send TaskStarted at t=1000
    TelemetryEvent started{TelemetryEventType::task_started, 1000, {}};
    sink_.enqueue(started);
    
    // Send FirstByteReceived at t=2000 (1000ms later)
    TelemetryEvent first_byte{TelemetryEventType::first_byte_received, 2000, {}};
    sink_.enqueue(first_byte);
    
    // Wait for processing
    std::this_thread::sleep_for(std::chrono::milliseconds(50));
    
    auto summary = collector_.final_summary();
    EXPECT_EQ(summary.time_to_first_byte_ms, 1); // 1000ms = 1ms in ns/1e6
}

TEST_F(TelemetryCollectorTest, TTFB_IgnoresDuplicate) {
    // Send TaskStarted at t=1000
    TelemetryEvent started{TelemetryEventType::task_started, 1000, {}};
    sink_.enqueue(started);
    
    // Send FirstByteReceived at t=2000
    TelemetryEvent first_byte1{TelemetryEventType::first_byte_received, 2000, {}};
    sink_.enqueue(first_byte1);
    
    // Send another FirstByteReceived at t=3000 (should be ignored)
    TelemetryEvent first_byte2{TelemetryEventType::first_byte_received, 3000, {}};
    sink_.enqueue(first_byte2);
    
    std::this_thread::sleep_for(std::chrono::milliseconds(50));
    
    auto summary = collector_.final_summary();
    // Should be 1ms (2000-1000), not 2ms (3000-1000)
    EXPECT_EQ(summary.time_to_first_byte_ms, 1);
}

TEST_F(TelemetryCollectorTest, PacketStatistics) {
    // Send TaskStarted
    TelemetryEvent started{TelemetryEventType::task_started, 1000, {}};
    sink_.enqueue(started);
    
    // Send DownloadDelta events with varying sizes
    TelemetryEvent delta1{TelemetryEventType::download_delta, 1100, {.download_delta = {100}}};
    TelemetryEvent delta2{TelemetryEventType::download_delta, 1200, {.download_delta = {200}}};
    TelemetryEvent delta3{TelemetryEventType::download_delta, 1300, {.download_delta = {150}}};
    sink_.enqueue(delta1);
    sink_.enqueue(delta2);
    sink_.enqueue(delta3);
    
    std::this_thread::sleep_for(std::chrono::milliseconds(50));
    
    auto summary = collector_.final_summary();
    EXPECT_EQ(summary.packets_enqueued_total, 3u);
    EXPECT_EQ(summary.max_packet_size_bytes, 200u);
}

TEST_F(TelemetryCollectorTest, MaxMemoryTracking) {
    TelemetryEvent started{TelemetryEventType::task_started, 1000, {}};
    sink_.enqueue(started);
    
    TelemetryEvent mem1{TelemetryEventType::memory_sample, 1100, {.memory_sample = {1000000}}};
    TelemetryEvent mem2{TelemetryEventType::memory_sample, 1200, {.memory_sample = {2000000}}};
    TelemetryEvent mem3{TelemetryEventType::memory_sample, 1300, {.memory_sample = {1500000}}};
    sink_.enqueue(mem1);
    sink_.enqueue(mem2);
    sink_.enqueue(mem3);
    
    std::this_thread::sleep_for(std::chrono::milliseconds(50));
    
    auto summary = collector_.final_summary();
    EXPECT_EQ(summary.max_memory_bytes, 2000000u);
}

TEST_F(TelemetryCollectorTest, PauseCounts) {
    TelemetryEvent started{TelemetryEventType::task_started, 1000, {}};
    sink_.enqueue(started);
    
    // QueuePaused events
    TelemetryEvent pause1{TelemetryEventType::queue_paused, 1100, {.queue_paused = {TelemetryPauseReason::queue_full}}};
    TelemetryEvent pause2{TelemetryEventType::queue_paused, 1200, {.queue_paused = {TelemetryPauseReason::gap}}};
    TelemetryEvent pause3{TelemetryEventType::queue_paused, 1300, {.queue_paused = {TelemetryPauseReason::queue_full}}};
    sink_.enqueue(pause1);
    sink_.enqueue(pause2);
    sink_.enqueue(pause3);
    
    std::this_thread::sleep_for(std::chrono::milliseconds(50));
    
    auto summary = collector_.final_summary();
    EXPECT_EQ(summary.total_pause_count, 3u);
    EXPECT_EQ(summary.queue_full_pause_count, 2u);
}

TEST_F(TelemetryCollectorTest, CurrentSnapshotHasWatermark) {
    TelemetryEvent started{TelemetryEventType::task_started, 1000, {}};
    sink_.enqueue(started);
    
    std::this_thread::sleep_for(std::chrono::milliseconds(10));
    
    auto snapshot = collector_.current_snapshot();
    EXPECT_GT(snapshot.watermark_timestamp_ns, 0u);
}

TEST_F(TelemetryCollectorTest, IgnoresEventsAfterTaskCompleted) {
    TelemetryEvent started{TelemetryEventType::task_started, 1000, {}};
    TelemetryEvent completed{TelemetryEventType::task_completed, 2000, {}};
    TelemetryEvent late_delta{TelemetryEventType::download_delta, 3000, {.download_delta = {100}}};
    
    sink_.enqueue(started);
    sink_.enqueue(completed);
    sink_.enqueue(late_delta); // Should be ignored
    
    std::this_thread::sleep_for(std::chrono::milliseconds(50));
    
    auto summary = collector_.final_summary();
    // Should still be 0 since delta after completion was ignored
    EXPECT_EQ(summary.packets_enqueued_total, 0u);
}

} // namespace asyncdownload::telemetry
