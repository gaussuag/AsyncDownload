Add case handler for memory_sample event:

```cpp
case TelemetryEventType::memory_sample: {
    size_t memory_bytes = event.payload.memory_sample.memory_bytes;
    
    // Update max_memory_bytes using CAS loop
    size_t current_max = max_memory_bytes_.load(std::memory_order_relaxed);
    while (memory_bytes > current_max) {
        if (max_memory_bytes_.compare_exchange_weak(current_max, memory_bytes,
                std::memory_order_relaxed, std::memory_order_relaxed)) {
            break;
        }
    }
    
    // Also update snapshot_ for current_snapshot() reading
    if (memory_bytes > snapshot_.memory_bytes) {
        snapshot_.memory_bytes = memory_bytes;
    }
    
    break;
}
```

The MemorySample payload provides `memory_bytes` field. Use compare_exchange_weak for lock-free max update.

Update the existing download_delta case handler to track inflight bytes.

The DownloadDelta payload provides `bytes` field and we need to track cumulative inflight. The inflight bytes should be accumulated as download_delta events come in (bytes received but not yet persisted).

Add to download_delta case:
```cpp
// Update inflight tracking
total_download_bytes_.fetch_add(bytes, std::memory_order_relaxed);
size_t current_inflight = total_download_bytes_.load(std::memory_order_relaxed) - 
                         total_persist_bytes_.load(std::memory_order_relaxed);

// Update max_inflight_bytes using CAS loop
size_t current_max = max_inflight_bytes_.load(std::memory_order_relaxed);
while (current_inflight > current_max) {
    if (max_inflight_bytes_.compare_exchange_weak(current_max, current_inflight,
            std::memory_order_relaxed, std::memory_order_relaxed)) {
        break;
    }
}

// Update snapshot_ for current_snapshot() reading
if (current_inflight > snapshot_.inflight_bytes) {
    snapshot_.inflight_bytes = current_inflight;
}
```

Note: total_persist_bytes_ is updated in persist_delta case. The inflight calculation is approximate (doesn't account for in-flight persist operations).
