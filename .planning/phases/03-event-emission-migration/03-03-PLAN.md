    Create tests/telemetry/telemetry_event_emission_test.cpp:
    ```cpp
    #include <gtest/gtest.h>
    #include "asyncdownload/telemetry.hpp"

    namespace asyncdownload::telemetry {

    TEST(TelemetryEventEmissionTest, FullDownloadSequence) {
        TelemetrySession session;

        session.record_task_started();
        std::this_thread::sleep_for(std::chrono::milliseconds(10));
        session.record_first_byte_received();

        for (int i = 0; i < 10; ++i) {
            session.record_download_delta(1024);
        }

        session.record_pause(TelemetryPauseReason::queue_full, true);
        session.record_pause(TelemetryPauseReason::memory_pressure, false);

        session.record_memory_sample(4096);

        for (int i = 0; i < 5; ++i) {
            session.record_persist_delta(1024);
        }

        session.record_task_completed();

        auto summary = session.final_summary();

        EXPECT_GT(summary.time_to_first_byte_ms, 0);
        EXPECT_GT(summary.packets_enqueued_total, 0);
        EXPECT_GE(summary.total_pause_count, 2);
        EXPECT_GE(summary.queue_full_pause_count, 1);
        EXPECT_GT(summary.max_memory_bytes, 0);
    }

    TEST(TelemetryEventEmissionTest, SnapshotDuringDownload) {
        TelemetrySession session;

        session.record_task_started();
        session.record_download_delta(2048);
        session.record_memory_sample(8192);

        auto snapshot = session.current_snapshot();

        EXPECT_GE(snapshot.downloaded_bytes, 2048);
        EXPECT_GE(snapshot.memory_bytes, 8192);
    }

    } // namespace asyncdownload::telemetry
    ```
