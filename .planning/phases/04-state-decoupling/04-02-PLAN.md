  <action>
    Refactor the invoke_progress function in src/download/download_engine.cpp to merge telemetry values from TelemetrySession with coordination values from SessionState/range state.

    The current invoke_progress function (around line 217) builds a ProgressSnapshot by:
    1. Reading downloaded_bytes, persisted_bytes from SessionState
    2. Calculating speeds via update_progress_rates (NOW REMOVED)
    3. Filling in range/handle state
    
    NEW approach:
    1. Get telemetry snapshot: `auto telemetry = session.telemetry_session_.current_snapshot();`
    2. Build complete snapshot by merging:
       - FROM TELEMETRY: downloaded_bytes, persisted_bytes, inflight_bytes, network_bytes_per_second, disk_bytes_per_second, memory_bytes, watermark_timestamp_ns
       - FROM SESSIONSTATE/RANGE: total_bytes, vdl_offset, queued_packets, resumed
       - FROM RANGE STATE: active_ranges, finished_ranges, paused_ranges, gap_paused_ranges, memory_paused_ranges
       - FROM HANDLE STATE: active_requests
    
    Example refactored invoke_progress structure:
    ```cpp
    void invoke_progress(core::SessionState& session,
                         const std::vector<std::unique_ptr<core::RangeContext>>& ranges,
                         const std::vector<TransferHandle>& handles) noexcept {
        if (!session.progress_callback) {
            return;
        }
        
        // Get telemetry-based values from TelemetrySession
        auto snapshot = session.telemetry_session_.current_snapshot();
        
        // Fill in coordination-based values
        snapshot.total_bytes = session.total_size;
        snapshot.vdl_offset = session.vdl_offset.load(std::memory_order_relaxed);
        snapshot.queued_packets = session.queued_packets.load(std::memory_order_relaxed);
        snapshot.resumed = session.resumed;
        
        // Range state
        for (const auto& range : ranges) {
            const auto status = static_cast<core::RangeStatus>(
                range->status.load(std::memory_order_acquire));
            if (status == core::RangeStatus::downloading || status == core::RangeStatus::paused) {
                ++snapshot.active_ranges;
            }
            if (range->marked_finished.load(std::memory_order_acquire)) {
                ++snapshot.finished_ranges;
            }
            if (status == core::RangeStatus::paused) {
                ++snapshot.paused_ranges;
            }
            if (range->pause_for_gap.load(std::memory_order_acquire)) {
                ++snapshot.gap_paused_ranges;
            }
            if (range->pause_for_memory.load(std::memory_order_acquire)) {
                ++snapshot.memory_paused_ranges;
            }
        }
        
        // Handle state
        snapshot.active_requests = static_cast<std::size_t>(std::count_if(handles.begin(),
            handles.end(),
            [](const TransferHandle& handle) {
                return handle.in_multi;
            }));
        
        try {
            session.progress_callback(snapshot);
        } catch (...) {
        }
    }
    ```
    
    NOTE: The memory_bytes from TelemetryCollector snapshot_ is already set via record_memory_sample calls in the telemetry system. The range loop no longer needs to track memory_bytes.
